= post_posting_code
:status: hidden

pass:q[`post_posting_code(postings, effective_date)`]  

Carries out actions after deciding on whether to accept a Posting Instruction Batch in the pass:q[`pre_posting_hook`].

This hook runs after the accepted Posting Instructions were inserted into the Postings Ledger.

For a product that wants to rebalance postings (for example, a product with credit lines), rebalancing would be done here.

[NOTE]
====
When a pass:q[`post_posting_code`] execution fails, it is marked as a failure in the database. Subsequent executions for that account or plan require this failure to be handled before they can continue. For more information, see link:{baseURL}/smart_contracts/contracts_api_3xx/best_practice_guidelines#behaviour_of_the_post_posting_code[Behaviour of the post_posting_code] and link:{baseURL}/api/core_api#post_posting_republisher[Post Posting Republisher].
====  

**Arguments**  

[width="100%", options="header"]
|===
|name|type|description
|postings|PostingInstructionBatch|The successful postings within PostingInstructionBatch.
|effective_date|datetime|The logical time the hook is being run.
|===

api = "3.12.0"
version = "1.0.0"
# Manual re-balance example (assuming all hard settlements)

parameters = [
    Parameter(
        name="balance_pot_to_type_map",
        description="Map of spend type (mcc) to balance pot (dict format - encoded json).",
        display_name="Map of spend type (mcc) to balance pot (dict format - encoded json).",
        level=Level.TEMPLATE,
        shape=StringShape,
        update_permission=UpdatePermission.USER_EDITABLE,
    )
]


@requires(parameters=True)
def post_posting_code(postings, effective_date):
    # Re-balance the default account address to appropriate
    # balance pots based on postings MCC codes
    balance_pot_to_mcc_map = vault.get_parameter_timeseries(name="balance_pot_to_type_map").latest()
    balance_pot_dict = json_loads(balance_pot_to_mcc_map)
    for posting in postings:
        initial_posting_net_balance = posting.balances()[
            (DEFAULT_ADDRESS, DEFAULT_ASSET, posting.denomination, Phase.COMMITTED)
        ].net
        # Need to re-balance this
        if posting.account_address == DEFAULT_ADDRESS:
            posting_mcc = posting.instruction_details.get("MCC", "")
            balance_pot = balance_pot_dict.get(posting_mcc, "PURCHASE")
            if posting.credit:
                # Credit to the account
                posting_ins = vault.make_internal_transfer_instructions(
                    amount=posting.amount,
                    denomination=posting.denomination,
                    from_account_id=vault.account_id,
                    from_account_address="",
                    to_account_id=vault.account_id,
                    to_account_address="CREDIT",
                    asset=posting.asset,
                    client_transaction_id="REBALANCE_{}".format(vault.account_id),
                    instruction_details={
                        "initial_posting_net_balance": str(initial_posting_net_balance)
                    },
                )
            else:
                posting_ins = vault.make_internal_transfer_instructions(
                    amount=posting.amount,
                    denomination=posting.denomination,
                    from_account_id=vault.account_id,
                    from_account_address=DEFAULT_ADDRESS,
                    to_account_id=vault.account_id,
                    to_account_address=balance_pot,
                    asset=posting.asset,
                    client_transaction_id="REBALANCE_{}".format(vault.account_id),
                    instruction_details={
                        "initial_posting_net_balance": str(initial_posting_net_balance)
                    },
                )
            vault.instruct_posting_batch(
                posting_instructions=posting_ins, effective_date=effective_date
            )
