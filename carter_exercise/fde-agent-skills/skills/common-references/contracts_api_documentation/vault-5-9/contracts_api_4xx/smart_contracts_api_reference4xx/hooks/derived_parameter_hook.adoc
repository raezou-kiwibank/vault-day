from decimal import Decimal
from contracts_api import (
    DerivedParameterHookArguments,
    DerivedParameterHookResult,
    NumberShape,
    Parameter,
    ParameterLevel,
    ParameterUpdatePermission,
    requires,
    Tside,
)

api = "4.0.0"
version = "1.0.0"
tside = Tside.ASSET

parameters = [
    Parameter(
        name="variable_interest_rate",
        shape=NumberShape(min_value=Decimal("0"), max_value=Decimal("1"), step=Decimal("0.0001")),
        level=ParameterLevel.TEMPLATE,
        description="The annual rate of the mortgage to be applied after the fixed rate term.",
        display_name="Variable interest rate (p.a.)",
        default_value=Decimal("0.129971"),
        update_permission=ParameterUpdatePermission.OPS_EDITABLE,
    ),
    Parameter(
        name="variable_rate_adjustment",
        shape=NumberShape(min_value=Decimal("-1"), max_value=Decimal("1"), step=Decimal("0.0001")),
        level=ParameterLevel.INSTANCE,
        description="Account level adjustment to be added to variable interest rate, "
        "can be positive, negative or zero.",
        display_name="Variable rate adjustment",
        default_value=Decimal("0.00"),
        update_permission=ParameterUpdatePermission.OPS_EDITABLE,
    ),
    Parameter(
        name="interest_rate_including_adjustment",
        shape=NumberShape(),
        level=ParameterLevel.INSTANCE,
        derived=True,
        display_name="Interest rate including adjustment",
        description="Annual interest rate including variable rate adjustment",
    ),
]

@requires(parameters=True)
def derived_parameter_hook(vault, hook_arguments: DerivedParameterHookArguments):
    interest_rate_including_adjustment = _get_interest_rate_including_adjustment(vault)

   return DerivedParameterHookResult(
        parameters_return_value={
            "interest_rate_including_adjustment": interest_rate_including_adjustment
        }
    )

def _get_interest_rate_including_adjustment(vault) -> Decimal:
    annual_rate = Decimal(vault.get_parameter_timeseries(name="variable_interest_rate").latest())
    annual_rate += Decimal(vault.get_parameter_timeseries(name="variable_rate_adjustment").latest())
    return annual_rate
