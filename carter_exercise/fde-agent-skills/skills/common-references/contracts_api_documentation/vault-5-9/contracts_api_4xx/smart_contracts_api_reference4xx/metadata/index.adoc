= Metadata

The metadata provides a description of the Smart Contract. Some of the metadata fields are
required in order for a Smart Contract to be valid.

The metadata consists of a docstring describing the Smart Contract, and a number of fields which
define important elements of it.  

[source,python]
----
"""
This is the description of the Smart Contract as a whole.
"""
api = '4.0.0' 
version = '1.2.3' # Labels are allowed, e.g., '1.2.3-alpha+build.1' 
expected_parameters = [ ExpectedParameter(id='denomination'), ExpectedParameter(id='transaction_code',constraint=StringConstraint()) ] 
global_parameters = ['denomination', 'central_bank_yearly_interest_rate'] 
parameters = [
  Parameter(
    name='late_payment_fee',
    shape=NumberShape(),
    level=ParameterLevel.TEMPLATE,
    description='Flat rate fee incurred when a payment becomes overdue',
    display_name='Late payment fee',
    default_value=Decimal('35.00')
  )
] 
tside = Tside.LIABILITY 
events_timezone = 'US/Pacific' 
supported_denominations = ['GBP', 'HKD'] 
event_types = [
    SmartContractEventType(
        name='ACCRUE_INTEREST'
    ),
    {%- if 'IMP_1774_FLEXIBLE_ADJUSTMENT_CORRECTIONS' in enabled_fflags %}
    SmartContractAdjustmentEventType(
        name='APPLY_INTEREST',
        correction_point=AdjustmentCorrectionPoint(
            value_datetime=DefinedDateTime.EFFECTIVE_DATETIME,
            booking_datetime=DefinedDateTime.EFFECTIVE_DATETIME,
        ),
    ),
    {%- else %}
    SmartContractEventType(
        name='APPLY_INTEREST', scheduler_tag_ids=['INTEREST']
    ),
    {%- endif %}
] 
event_types_groups = [
    EventTypesGroup(
        name='GROUP_INTEREST',
        event_types_order=['ACCRUE_INTEREST', 'APPLY_INTEREST']
    ),
] 
data_fetchers = [
    BalancesIntervalFetcher(
        fetcher_id='bif_1',
        start=RelativeDateTime(
            shift=Shift(years=-1, months=2),
            origin=DefinedDateTime.EFFECTIVE_DATETIME
        ),
        end=DefinedDateTime.EFFECTIVE_DATETIME
    ),
    BalancesObservationFetcher(
        fetcher_id='bof_1',
        at=RelativeDateTime(
            shift=Shift(months=-1),
            origin=DefinedDateTime.EFFECTIVE_DATETIME
        ),
        filter=BalancesFilter(addresses=['INTEREST', 'SAVINGS'])
    ),
    BalancesObservationFetcher(
        fetcher_id='lbof_1',
        at=DefinedDateTime.LIVE,
    ),
    PostingsIntervalFetcher(
        fetcher_id='pif_1',
        start=RelativeDateTime(
            shift=Shift(years=-1, months=2),
            origin=DefinedDateTime.EFFECTIVE_DATETIME
        ),
    )
] 
notification_types = ['statement_created', 'repayment_made'] 
address_details = [
  AddressDetails(
    account_address='DEFAULT',
    description='The default address',
    tags=['default'],
  ),
  AddressDetails(
    account_address='INTEREST',
    description='Keeping track of interest accrued',
    tags=['interest'],
  ),
  AddressDetails(
    account_address='FEES',
    description='Keeping track of fees charged',
    tags=['fees'],
  ),
] 
attributes = [
  Attribute(name='accounting_balance', data_type=AttributeDecimalType()),
  Attribute(name='last_interest_accrual_date', data_type=AttributeDateTimeType()),
] 
adjustment_strategy = AdjustmentStrategy.SCHEDULE_TRIGGERED
