= attribute_hook
:status: hidden

pass:q[`attribute_hook(vault, hook_arguments)`]  

Returns the calculated value for a specific Attribute at an effective time.

This hook runs when pass:q[`GET /v1/account-attribute-values`] is called with one or more matching pass:q[`account_ids`] and pass:q[`attribute_names`].

Note: Account Data or Hook Requirements can be defined to be fetched for a specific attribute only by setting the pass:q[`attribute_name`] on the link:{baseURL}/smart_contracts/contracts_api_4xx/common_types_4xx/decorators#fetch_account_data[fetch_account_data]{empty} or link:{baseURL}/smart_contracts/contracts_api_4xx/common_types_4xx/decorators#requires[requires]{empty} decorators.

from contracts_api import (
    Attribute,
    AttributeDecimalType,
    AttributeHookResult,
    AttributeStringType,
    BalancesObservationFetcher,
    DefinedDateTime,
    fetch_account_data,
    Phase,
    requires,
)

api = "4.0.0"
version = "1.0.0"

attributes = [
    Attribute(
        name="current_balance",
        data_type=AttributeDecimalType(),
    ),
    Attribute(
        name="latest_flag",
        data_type=AttributeStringType(),
    ),
]

data_fetchers = [
    BalancesObservationFetcher(
        fetcher_id="effective_balance",
        at=DefinedDateTime.EFFECTIVE_DATETIME,
    ),
]

COMMITTED_DEFAULT_BALANCE_ADDRESS = ("DEFAULT", "COMMERCIAL_BANK_MONEY", "GBP", Phase.COMMITTED)


@fetch_account_data(
    attribute_name="current_balance",
    balances=["effective_balance"],
)
@requires(attribute_name="latest_flag", flags=True)
def attribute_hook(vault, hook_arguments):
    if hook_arguments.attribute_name == "current_balance":
        balances = vault.get_balances_observation(fetcher_id="effective_balance").balances
        committed_effective_balances = balances.get(COMMITTED_DEFAULT_BALANCE_ADDRESS)
        return AttributeHookResult(attribute_value=committed_effective_balances.net)
    elif hook_arguments.attribute_name == "latest_flag":
        flag_enabled = vault.get_flag_timeseries(flag="flag-definition-id").latest()
        if flag_enabled:
            return AttributeHookResult(attribute_value="Flag enabled")
        return AttributeHookResult(attribute_value="Flag disabled")
    return AttributeHookResult()
