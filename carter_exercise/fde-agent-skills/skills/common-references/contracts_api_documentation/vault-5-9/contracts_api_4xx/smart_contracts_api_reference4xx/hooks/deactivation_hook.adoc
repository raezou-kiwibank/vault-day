= deactivation_hook
:status: hidden

pass:q[`deactivation_hook(vault, hook_arguments)`]  

Carries out actions when the account closes.

This hook runs after a request to set a Customer Account to pass:q[`ACCOUNT_STATUS_CLOSED`], either via the v1 or v2 Accounts endpoints.

Note: Ensure all balances are 0 before returning from this hook.

from decimal import Decimal
from contracts_api import (
    BalanceCoordinate,
    BalancesObservationFetcher,
    DeactivationHookArguments,
    DeactivationHookResult,
    DEFAULT_ADDRESS,
    DEFAULT_ASSET,
    DefinedDateTime,
    fetch_account_data,
    Phase,
    Rejection,
    RejectionReason,
    Tside,
)

api = "4.0.0"
version = "1.0.0"
tside = Tside.ASSET

data_fetchers = [
    BalancesObservationFetcher(
        fetcher_id="live_balances",
        at=DefinedDateTime.LIVE,
    ),
]

@fetch_account_data(balances=["live_balances"])
def deactivation_hook(vault, hook_arguments: DeactivationHookArguments):
    outstanding_debt = _get_all_outstanding_debt(vault)
    if outstanding_debt != Decimal("0"):
        return DeactivationHookResult(
            rejection=Rejection(
                message=f"This account has {outstanding_debt} outstanding."
                "Cannot close account until account balance nets to 0",
                reason_code=RejectionReason.AGAINST_TNC,
            )
        )

def _get_all_outstanding_debt(vault) -> Decimal:
    addresses = [DEFAULT_ADDRESS, "ACCRUED_INTEREST"]
    balances = vault.get_balances_observation(fetcher_id="live_balances").balances

   return Decimal(
        sum(
            balances[BalanceCoordinate(address, DEFAULT_ASSET, "GBP", Phase.COMMITTED)].net
            for address in addresses
        )
    )
