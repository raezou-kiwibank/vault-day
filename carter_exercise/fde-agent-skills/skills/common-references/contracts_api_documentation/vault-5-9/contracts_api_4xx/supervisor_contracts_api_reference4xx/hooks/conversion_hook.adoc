= conversion_hook
:status: hidden

pass:q[`conversion_hook(vault, hook_arguments)`]  

Carries out actions after a new contract has been activated on this plan - the contract template is upgraded to a new version.

If pass:q[`event_types`] are defined on a Supervisor Contract, then this hook must be defined in order to convert a Plan to that Supervisor Contract.

from contracts_api import (
    ScheduledEvent, ScheduleExpression, SmartContractDescriptor, SupervisorContractEventType,
    SupervisorConversionHookResult,
)

api = "4.0.0"
version = "1.0.0"

supervised_smart_contracts = [
    SmartContractDescriptor(
        alias="current_account",
        smart_contract_version_id="1",
    )
]

event_types = [
    SupervisorContractEventType(
        name="SUPERVISED_SERVICE_CHARGE",
        overrides_event_types=[("current_account", "SERVICE_CHARGE")],
    ),
    SupervisorContractEventType(
        name="SUPERVISED_OVERDRAFT_FEE",
        overrides_event_types=[("current_account", "OVERDRAFT_FEE")],
    ),
    SupervisorContractEventType(name="SUPERVISOR_MONTHLY_REPORT"),
]


def conversion_hook(vault, hook_arguments):
    effective_datetime = hook_arguments.effective_datetime
    return SupervisorConversionHookResult(
        scheduled_events_return_value={
            "SUPERVISED_SERVICE_CHARGE": hook_arguments.existing_schedules[
                "SUPERVISED_SERVICE_CHARGE"
            ],
            "SUPERVISED_OVERDRAFT_FEE": hook_arguments.existing_schedules[
                "SUPERVISED_OVERDRAFT_FEE"
            ],
            "SUPERVISOR_MONTHLY_REPORT": ScheduledEvent(
                start_datetime=effective_datetime,
                expression=ScheduleExpression(day="*"),
            ),
        },
    )
