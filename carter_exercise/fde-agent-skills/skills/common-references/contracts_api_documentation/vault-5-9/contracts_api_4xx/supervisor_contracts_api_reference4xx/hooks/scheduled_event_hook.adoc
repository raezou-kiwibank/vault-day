from decimal import Decimal
from contracts_api import (
    CustomInstruction,
    PostingInstructionsDirective,
    requires,
    ScheduledEvent,
    ScheduleExpression,
    SmartContractDescriptor,
    SupervisorActivationHookResult,
    SupervisorContractEventType,
    SupervisorScheduledEventHookResult,
)

api = "4.0.0"
version = "1.0.0"

supervised_smart_contracts = [
    SmartContractDescriptor(
        alias="current_account",
        smart_contract_version_id="current_account_v1",
    )
]

event_types = [
    SupervisorContractEventType(
        name="SUPERVISED_OVERDRAFT_FEE",
        overrides_event_types=[("current_account", "OVERDRAFT_FEE")],
    ),
]


def activation_hook(vault, hook_arguments):
    return SupervisorActivationHookResult(
        scheduled_events_return_value={
            "SUPERVISED_OVERDRAFT_FEE": ScheduledEvent(
                start_datetime=vault.get_plan_opening_datetime(),
                expression=ScheduleExpression(hour=0, minute=0, second=0),
            ),
        },
    )


@requires(event_type="SUPERVISED_OVERDRAFT_FEE", supervisee_hook_directives="all")
def scheduled_event_hook(vault, hook_arguments):
    supervisee_posting_instructions_directives = {}
    if hook_arguments.event_type == "SUPERVISED_OVERDRAFT_FEE":
        for account_id, supervisee in vault.supervisees.items():
            product = supervisee.get_alias()
            if product != "current_account":
                continue
            supervisee_hook_result = supervisee.get_hook_result()
            for directive in supervisee_hook_result.posting_instructions_directives:
                overdraft_fee_instructions = (
                    instruction
                    for instruction in directive.posting_instructions
                    if "OVERDRAFT_FEE" in instruction.instruction_details
                )
            for instruction in overdraft_fee_instructions:
                for posting in instruction.postings:
                    posting.amount = posting.amount * Decimal("0.5")
                posting_ins = CustomInstruction(
                    instruction_details=instruction.instruction_details,
                    postings=instruction.postings,
                )
                supervisee_posting_instructions_directives[account_id] = [
                    PostingInstructionsDirective(
                        value_datetime=instruction.value_datetime,
                        client_batch_id=instruction.client_batch_id,
                        posting_instructions=[posting_ins],
                    )
                ]
    if not supervisee_posting_instructions_directives:
        return
    return SupervisorScheduledEventHookResult(
        supervisee_posting_instructions_directives=supervisee_posting_instructions_directives
    )
