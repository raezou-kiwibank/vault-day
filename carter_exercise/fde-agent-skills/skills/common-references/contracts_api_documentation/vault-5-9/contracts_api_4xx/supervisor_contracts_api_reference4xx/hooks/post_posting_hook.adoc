= post_posting_hook
:status: hidden

pass:q[`post_posting_hook(vault, hook_arguments)`]  

Carries out actions after a supervisee has decided whether to accept a batch of posting instructions in its pass:q[`pre_posting_hook`] hook.

This hook is called after the accepted posting instructions were inserted into the Vault posting ledger.

from dateutil.relativedelta import relativedelta
from contracts_api import (
    BalanceCoordinate, BalanceDefaultDict, DEFAULT_ADDRESS, DEFAULT_ASSET, Phase, requires,
    ScheduleExpression, SmartContractDescriptor, SupervisorContractEventType,
    SupervisorPostPostingHookResult, UpdatePlanEventTypeDirective,
)

api = "4.0.0"
version = "1.0.0"

supervised_smart_contracts = [
    SmartContractDescriptor(
        alias="current_account",
        smart_contract_version_id="current_account_v1",
        supervise_post_posting_hook=True,
    ),
    SmartContractDescriptor(
        alias="savings_account",
        smart_contract_version_id="savings_account_v1",
    ),
]

event_types = [
    SupervisorContractEventType(name="APPLY_OVERDRAFT_FEE"),
]


@requires(balances="latest live", data_scope="all", parameters=True)
def post_posting_hook(vault, hook_arguments):
    effective_datetime = hook_arguments.effective_datetime
    overdraft_limit = None
    grace_period = 0
    current_balance = 0
    savings_balance = 0
    balance_update = 0
    for account_id, supervisee in vault.supervisees.items():
        balances = {}
        denomination = supervisee.get_parameter_timeseries(name="denomination").latest()
        balance_key = BalanceCoordinate(
            DEFAULT_ADDRESS,
            DEFAULT_ASSET,
            denomination,
            Phase.COMMITTED,
        )
        product = supervisee.get_alias()
        if product == "current_account":
            current_balance = supervisee.get_balances_timeseries()[balance_key].latest().net
            posting_instructions = hook_arguments.supervisee_posting_instructions[account_id]
            balances.update(
                {
                    account_id: sum(
                        (pi.balances() for pi in posting_instructions),
                        BalanceDefaultDict(),
                    )
                }
            )
            balance_update = balances[account_id][balance_key].net
            overdraft_limit = supervisee.get_parameter_timeseries(name="overdraft_limit").latest()
            grace_period = supervisee.get_parameter_timeseries(name="grace_period").latest()
        elif product == "savings_account":
            savings_balance = supervisee.get_balances_timeseries()[balance_key].latest().net
    if overdraft_limit is None or balance_update == 0:
        return
    if current_balance + savings_balance + balance_update < -overdraft_limit:
        overdraft_fee_date = effective_datetime + relativedelta(days=int(grace_period))
        update_plan_event_type_directive = UpdatePlanEventTypeDirective(
            event_type="APPLY_OVERDRAFT_FEE",
            expression=ScheduleExpression(
                day=overdraft_fee_date.day,
            ),
            skip=False,
        )
        return SupervisorPostPostingHookResult(
            update_plan_event_type_directives=[update_plan_event_type_directive]
        )
